Q1. What are decision-making (conditional) statements in C?

Answer:

Decision-making statements allow execution of different code blocks based on conditions.

Types of Conditional Statements:

  • if - Execute block if condition is true
  • if-else - Execute one of two blocks
  • else-if ladder - Multiple conditions
  • nested if - if inside another if
  • switch - Select from multiple options

Q2. Explain the if statement with syntax and example.

Answer:

Syntax:

Syntaxif (condition) {
    // code to execute if condition is true
}

Example:

C#include <stdio.h>

int main() {
    int age = 20;
    
    if (age >= 18) {
        printf("You are eligible to vote.\n");
    }
    
    return 0;
}

Flowchart:

flowchart TD A([START]) --> B{Condition?} B -->|True| C[Execute Statement] B -->|False| D[Skip] C --> E([END]) D --> E

Q3. Explain the if-else statement with syntax and example.

Answer:

Syntax:

Syntaxif (condition) {
    // code if condition is true
} 
else {
    // code if condition is false
}

Example:

C#include <stdio.h>

int main() {
    int num;
    
    printf("Enter a number: ");
    scanf("%d", &num);
    
    if (num % 2 == 0) {
        printf("%d is even.\n", num);
    } 
    else {
        printf("%d is odd.\n", num);
    }
    
    return 0;
}

Q5. Explain the switch statement with syntax and example.

Answer:

Syntax:

Syntaxswitch (expression) {
    case value1:

Q9. Explain the for loop with syntax and example.

Answer:

Syntax:

Syntaxfor (initialization; condition; update) {
    // loop body
}

Example - Print 1 to 10:

C#include <stdio.h>

int main() {
    int i;
    
    printf("Numbers 1 to 10:\n");

    }
    
    printf("\n");
    
    return 0;
}

Q10. Explain the while loop with syntax and example.

Answer:

Syntax:

Syntaxwhile (condition) {
    // loop body
    // update statement
}

Example - Sum of digits:

C#include <stdio.h>

int main() {
    int num, sum = 0, digit;
    
    printf("Enter a number: ");
    scanf("%d", &num);
    
    while (num > 0) {
        digit = num % 10;     // Get last digit
        sum = sum + digit;     // Add to sum
        num = num / 10;        // Remove last digit
    }
    
    printf("Sum of digits: %d\n", sum);
    
    return 0;
}

Q11. Explain the do-while loop with syntax and example.

Answer:

Syntax:

Syntaxdo {
    // loop body
} while (condition);

Key Point:

Executes at least once, even if condition is false.

Example - Menu-driven program:

C#include <stdio.h>

int main() {
    int choice;
    
    do {
        printf("\n=== MENU ===\n");
        printf("1. Option 1\n");
        printf("2. Option 2\n");
        printf("3. Exit\n");
        printf("Enter choice: ");
        scanf("%d", &choice);
        
        switch (choice) {
            case 1:
                printf("You selected Option 1\n");
                break;
            case 2:
                printf("You selected Option 2\n");
                break;
            case 3:
                printf("Exiting...\n");
                break;
            default:
                printf("Invalid choice!\n");
        }
    } while (choice != 3);
    
    return 0;
}

Q13-14. Explain break and continue statements.

Answer:

break Statement:

  • Immediately exits the loop
  • Transfers control to the statement after the loop
  • Used in loops and switch

continue Statement:

  • Skips current iteration
  • Jumps to next iteration
  • Used only in loops

Example:

C#include <stdio.h>

int main() {
    int i;
    
    // break example - stops at 5
    printf("Break example: ");
    for (i = 1; i <= 10; i++) {
        if (i == 5) {
            break;  // Exit loop
        }
        printf("%d ", i);
    }
    // Output: 1 2 3 4
    
    printf("\n");
    
    // continue example - skips 5
    printf("Continue example: ");
    for (i = 1; i <= 10; i++) {
        if (i == 5) {
            continue;  // Skip this iteration
        }
        printf("%d ", i);
    }
    // Output: 1 2 3 4 6 7 8 9 10
    
    return 0;
}

Q15-16. Write programs for factorial and Fibonacci series.

Answer:

Program 1 - Factorial:

C#include <stdio.h>

int main() {
    int n, i;
    long factorial = 1;
    
    printf("Enter a number: ");
    scanf("%d", &n);
    
    for (i = 1; i <= n; i++) {
        factorial = factorial * i;
    }
    
    printf("Factorial of %d = %ld\n", n, factorial);
    
    return 0;
}

Program 2 - Fibonacci Series:

C#include <stdio.h>

int main() {
    int n, i;
    int first = 0, second = 1, next;
    
    printf("Enter number of terms: ");
    scanf("%d", &n);
    
    printf("Fibonacci Series: ");
    
    for (i = 0; i < n; i++) {
        if (i <= 1) {
            next = i;
        } 
        else {
            next = first + second;
            first = second;
            second = next;
        }
        printf("%d ", next);
    }
    
    printf("\n");
    
    return 0;
}

Q21-23. Prime Number, Palindrome, and Armstrong Programs

Answer:

Program 1 - Check Prime Number:

C#include <stdio.h>

int main() {
    int num, i, isPrime = 1;
    
    printf("Enter a number: ");
    scanf("%d", &num);
    
    if (num <= 1) {
        isPrime = 0;
    } 
    else {
        for (i = 2; i * i <= num; i++) {
            if (num % i == 0) {
                isPrime = 0;
                break;
            }
        }
    }
    
    if (isPrime) {
        printf("%d is a Prime number.\n", num);
    } 
    else {
        printf("%d is not a Prime number.\n", num);
    }
    
    return 0;
}

Program 2 - Check Palindrome:

C#include <stdio.h>

int main() {
    int num, original, reversed = 0, digit;
    
    printf("Enter a number: ");
    scanf("%d", &num);
    
    original = num;
    
    while (num > 0) {
        digit = num % 10;
        reversed = reversed * 10 + digit;
        num = num / 10;
    }
    
    if (original == reversed) {
        printf("%d is a Palindrome.\n", original);
    } 
    else {
        printf("%d is not a Palindrome.\n", original);
    }
    
    return 0;
}

Program 3 - Check Armstrong Number:

C#include <stdio.h>
#include <math.h>

int main() {
    int num, original, digits = 0, sum = 0, digit;
    
    printf("Enter a number: ");
    scanf("%d", &num);
    
    original = num;
    
    // Count digits
    while (num > 0) {
        digits++;
        num = num / 10;
    }
    
    num = original;
    
    // Calculate sum of powers
    while (num > 0) {
        digit = num % 10;
        sum = sum + pow(digit, digits);
        num = num / 10;
    }
    
    if (sum == original) {
        printf("%d is an Armstrong number.\n", original);
    } 
    else {
        printf("%d is not an Armstrong number.\n", original);
    }
    
    return 0;
}

Q24-26. Calculator, goto Statement, and Nested Loops

Answer:

Simple Calculator:

C#include <stdio.h>

int main() {
    char operator;
    float num1, num2, result;
    
    printf("Enter operator (+, -, *, /): ");
    scanf("%c", &operator);
    
    printf("Enter two numbers: ");
    scanf("%f %f", &num1, &num2);
    
    switch (operator) {
        case '+':
            result = num1 + num2;
            break;
        case '-':
            result = num1 - num2;
            break;
        case '*':
            result = num1 * num2;
            break;
        case '/':
            if (num2 != 0) {
                result = num1 / num2;
            } 
            else {
                printf("Error: Division by zero!\n");
                return 1;
            }
            break;
        default:
            printf("Invalid operator!\n");
            return 1;
    }
    
    printf("Result: %.2f\n", result);
    
    return 0;
}

goto Statement (Use Sparingly!):

C#include <stdio.h>

int main() {

loop:
    printf("%d ", i);
    i++;
    
    if (i <= 5) {
        goto loop;  // Jump back to loop label
    }
    
    printf("\n");

}
